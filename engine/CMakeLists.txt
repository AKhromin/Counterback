add_library(counterback_engine src/matching_engine.cpp)
add_library(counterback::engine ALIAS counterback_engine)
target_include_directories(counterback_engine PUBLIC include)
target_compile_features(counterback_engine PUBLIC cxx_std_17)
set_target_properties(counterback_engine PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(counterback_engine PRIVATE /W4 /WX)
else()
    target_compile_options(counterback_engine PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(counterback_demo examples/matching_demo.cpp)
target_link_libraries(counterback_demo PRIVATE counterback::engine)

    add_test(NAME matching_engine COMMAND counterback_engine_tests)
endif()
